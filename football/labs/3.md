## часть 3. range partitioning по числовому значению
```sql
create table products (
    id bigint not null,
    name text not null,
    price numeric not null
) partition by range (price);

create table products_cheap
partition of products
for values from (0) to (100);

create table products_medium
partition of products
for values from (100) to (1000);

create table products_expensive
partition of products
for values from (1000) to (maxvalue);
```

## часть 4. list partitioning
```sql
create table customers (
    id bigint not null,
    name text not null,
    customer_type varchar(30) not null
) partition by list (customer_type);

create table customers_b2c
partition of customers
for values in ('B2C');

create table customers_b2b
partition of customers
for values in ('B2B');

create table customers_enterprise
partition of customers
for values in ('Enterprise');

insert into customers
select g,
       'клиент ' || g,
       case when r < 0.7 then 'B2C' when r < 0.95 then 'B2B' else 'Enterprise' end
from (select g, random() as r from generate_series(1, 300000) g) s;
```
```
    partition_name    | count  
----------------------+--------
 customers_b2c        | 209904
 customers_b2b        |  75104
 customers_enterprise |  14992
```
```sql
explain (analyze, buffers)
select *
from customers
where customer_type = 'B2B';
```
```
Seq Scan on customers_b2b customers  (cost=0.00..1491.80 rows=75104 width=31) (actual time=0.006..7.322 rows=75104 loops=1)
  Filter: ((customer_type)::text = 'B2B'::text)
  Buffers: shared hit=553
Planning Time: 0.068 ms
Execution Time: 10.039 ms
```
* какая партиция используется: только customers_b2b, 553 страницы
* почему постгрес может не обращаться к другим: у каждой партиции записан свой список значений, и постгрес знает что 'B2B' может лежать только в customers_b2b, в b2c и enterprise таких строк быть не может по определению
* чем отличается от range: range режет упорядоченные значения на отрезки (даты, цены), партиция это диапазон от и до. list перечисляет конкретные значения без всякого порядка (типы, страны, статусы), партиция это список. новое значение в range само попадет в подходящий отрезок, а в list для него нужна своя партиция или default

## часть 5. list partitioning и неизвестное значение
```sql
insert into customers values (1000001, 'Test User', 'VIP');
```
```
ERROR:  no partition of relation "customers" found for row
DETAIL:  Partition key of the failing row contains (customer_type) = (VIP).
```
(id взял 1000001 а не 100, потому что 100 уже занят сгенеренными клиентами)
```sql
create table customers_default
partition of customers default;

insert into customers values (1000001, 'Test User', 'VIP')
returning id, customer_type, tableoid::regclass as partition_name;
```
```
   id    | customer_type |  partition_name   
---------+---------------+-------------------
 1000001 | VIP           | customers_default
```
запрос по VIP теперь читает только default:
```
Seq Scan on customers_default customers  (cost=0.00..16.88 rows=3 width=118) (actual time=0.002..0.002 rows=1 loops=1)
  Filter: ((customer_type)::text = 'VIP'::text)
  Buffers: shared hit=1
Planning Time: 0.022 ms
Execution Time: 0.005 ms
```
и еще проверил что будет если потом захотеть сделать нормальную партицию для VIP:
```sql
create table customers_vip
partition of customers
for values in ('VIP');
```
```
ERROR:  updated partition constraint for default partition "customers_default" would be violated by some row
```
1) зачем нужна default partition? чтобы строки со значением для которого нет своей партиции не падали с ошибкой а складывались в запасную партицию
2) чем она полезна? приложение не ломается когда появилось новое значение: новый тип клиента, новая страна, опечатка. вставка проходит, данные не теряются
3) какие проблемы могут возникнуть если постоянно складывать новые значения в default?
* default превращается в свалку, в ней вперемешку все незнакомые значения, и запрос по любому из них читает всю default целиком, pruning тут уже не помогает
* ошибки маскируются: опечатка 'b2b' вместо 'B2B' молча уедет в default и никто не заметит
* нельзя спокойно выделить новую партицию: при create table ... partition of ... for values in ('VIP') постгрес проверяет что в default нет строк с VIP, и если есть то падает (у меня так и вышло). приходится сначала переносить строки из default руками, а на большой default эта проверка еще и долгая и держит блокировку

2) почему hash может быть полезен? когда надо просто равномерно размазать данные, а естественных диапазонов или категорий нет. каждая партиция и ее индексы в 4 раза меньше, vacuum идет по частям, а запрос по одному user_id читает 1 партицию из 4 (у меня user_events_2, 3676 страниц)
3) чем hash отличается от range? range держит соседние значения вместе и режет по диапазонам, поэтому отсекает партиции и для условий больше, меньше, between. hash специально перемешивает значения, соседние user_id лежат в разных партициях, поэтому отсечение работает только для равенства. between 1 and 100 читает все 4 партиции (план выше, parallel append по всем, 14708 страниц)
4) почему hash плохо подходит для сценария «удалить данные старше 3 лет»? в каждой hash партиции лежат данные за все даты вперемешку, старую партицию целиком не удалишь. остается только delete where created_at < ... по всем партициям: это миллионы строк, пишется wal, таблица пухнет и потом нужен vacuum. в range по месяцам это просто drop table старой партиции за миллисекунды

## часть 7. выбор стратегии партиционирования
* сценарий a: range по created_at (по дням или месяцам). старые данные лежат в своих партициях и через 3 года удаляются drop table или detach целой партицией, без delete миллионов строк
* сценарий b: list по customer_type. 3 фиксированных значения и запросы фильтруют как раз по ним, значит запрос по типу читает одну партицию
* сценарий c: hash по user_id. нужна именно равномерность, а у идентификатора нет естественных диапазонов
* сценарий d: range по created_at. аналитика всегда с between по дате, значит pruning оставит только партиции нужного периода
* сценарий e: list по стране. конечный набор кодов (`EE`, `LV`, `LT`, `FI`, `SE`), в одну партицию можно положить несколько стран, а на новую страну сделать default

как шаги из методички легли на код:
1) определить сегодняшнюю дату: date.today(), а для проверки можно передать `--today`
2) определить горизонт: сегодня и еще ahead шагов вперед, для 3 дней это 4 партиции
3) получить существующие партиции: из системного каталога pg_inherits
4) найти отсутствующие: сравниваю имена нужных и существующих
5) создать их: create table ... partition of ...

sql в репозитории:
```python
def list_partitions(conn, table: str) -> list[str]:
    with conn.cursor() as cur:
        cur.execute(
            """
            select c.relname as name
            from pg_inherits i
            join pg_class c on c.oid = i.inhrelid
            where i.inhparent = %s::regclass
            order by c.relname
            """,
            (table,),
        )
        return [row["name"] for row in cur.fetchall()]


def create_partition(conn, table: str, name: str, start: str, end: str) -> None:
    schema = table.split(".")[:-1]
    with conn.cursor() as cur:
        cur.execute(
            sql.SQL(
                "create table {} partition of {} for values from ({}) to ({})"
            ).format(
                sql.Identifier(*schema, name),
                table_identifier(table),
                sql.Literal(start),
                sql.Literal(end),
            )
        )
```
логика job в сервисе:
```python
def required_partitions(table: str, step: str, ahead: int, today: date):
    first = today if step == "day" else today.replace(day=1)
    base = table.split(".")[-1]
    pattern = "%Y_%m_%d" if step == "day" else "%Y_%m"
    result = []
    for i in range(ahead + 1):
        start = shift(first, step, i)
        result.append((f"{base}_{start.strftime(pattern)}", start, shift(first, step, i + 1)))
    return result


def create_partitions(table: str, step: str, ahead: int, today: date | None = None) -> bool:
    today = today or date.today()
    log(f"{datetime.now():%Y-%m-%d %H:%M} partition job started")

    with db.connection() as conn:
        existing = set(repo.list_partitions(conn, table))
    required = required_partitions(table, step, ahead, today)
    missing = [item for item in required if item[0] not in existing]

    log(f"table: {table}, today: {today.isoformat()}, horizon: {ahead} {STEP_WORDS[step]}")
    log(f"existing partitions: {len(existing)}")
    log(f"required partitions: {len(required)}")
    log(f"missing partitions: {len(missing)}")

    failed = 0
    for name, start, end in missing:
        log(f"creating: {name}")
        try:
            with db.connection() as conn:
                repo.create_partition(conn, table, name, bound(start), bound(end))
            log("partition created successfully")
        except psycopg.errors.DuplicateTable:
            log("partition already exists, skipped")
        except psycopg.Error as exc:
            failed += 1
            log(f"partition was not created: {exc}")

    log("partition job finished")
    return failed == 0
```
требования:
* создавать недостающие партиции: создаются только те из нужных которых нет среди существующих
* не создавать уже существующие: они отсеиваются заранее. а если партицию успел создать кто то другой между проверкой и созданием, постгрес кидает `DuplicateTable`, job пишет partition already exists, skipped и идет дальше. такая гонка у меня реально была: при первом запуске scheduler стартанул в тот же момент когда backend применял миграцию 004 и создавал те же партиции. отдельно проверил этот случай, вызвал создание уже существующей events_2026_09_14 и получил `DuplicateTable caught: relation "events_2026_09_14" already exists`
* быть безопасной при повторном запуске: второй запуск видит что все есть и ничего не делает (лог ниже)
* логировать результат: каждый шаг пишется в лог контейнера в формате как в методичке

## часть 11. production сценарий: ночной job сломался
`PartitionHealthCheck` это app/jobs/check_partitions.py. она берет те же нужные партиции что и job (сегодня плюс горизонт), помечает есть ✓ или нет ✗, и если чего то не хватает результат CRITICAL. код выхода 0 если OK и 2 если CRITICAL

```python
def check_partitions(table: str, step: str, ahead: int, today: date | None = None):
    today = today or date.today()
    with db.connection() as conn:
        existing = set(repo.list_partitions(conn, table))

    log(f"{datetime.now():%Y-%m-%d %H:%M} partition check: {table}")
    missing = []
    for name, start, _ in required_partitions(table, step, ahead, today):
        mark = "✓" if name in existing else "✗"
        log(f"{start.isoformat()}  {mark}  {name}")
        if name not in existing:
```

```python
    checked_at = datetime.now()
```

## часть 12. партиционирование собственной базы данных

### шаг 1. таблица
match_events. сейчас 5 000 029 строк, 417 MB данных и 467 MB индексов (замер из лабы 2 перед партиционированием). это самая большая и быстрее всех растущая таблица сервиса, события пишутся по времени и после матча почти не меняются

### шаг 2. ключ
created_at
* по нему фильтруют `GET /api/events?from=...&to=...` и отчет `GET /api/reports/top-scorers?from=...&to=...`, и сортировка списка событий по умолчанию тоже по нему
* данные копятся по времени: свежие месяцы читают чаще, а старые сезоны когда нибудь можно унести в архив целой партицией
* created_at у события не меняется (update_event меняет только event_type, minute и updated_at), значит строка никогда не будет переезжать в другую партицию

что еще думал: hash по match_id дал бы равномерность, но запросы по датам читали бы все партиции. list по event_type всего 5 значений и они неравномерные (замены 35%, красные 5%), и для дат тоже без пользы

### шаг 3. стратегия
range по месяцам. в месяце около 210 тыс событий, одна партиция весит 32 ... 37 MB (список ниже). по дням было бы по 7 тыс строк в партиции и больше 700 паритиций за 2 года, это лишняя нагрузка на планировщик. по годам по 2.5м строк в партиции, и отчет за месяц читал бы весь год

### шаг 4. какие запросы должны получить пользу от partition pruning
```
запрос                                                  | pruning | почему
GET /api/events?from=...&to=... (count и страница)      | да      | условие на created_at
GET /api/reports/top-scorers?from=...&to=...            | да      | условие на created_at
GET /api/events/{id}                                    | нет     | по id не понять месяц, pkey ищется в каждой партиции
GET /api/events?player_id=5000                          | нет     | даты нет, индекс по игроку в каждой партиции
GET /api/events без фильтров                            | нет     | count идет по всем партициям
```

### шаг 5. партиции
миграция sql/migrations/004_partition_match_events.sql:
```sql
alter table match_events rename to match_events_old;
alter index match_events_pkey rename to match_events_old_pkey;
alter index idx_match_events_player_created_at rename to idx_match_events_old_player_created_at;
alter index idx_match_events_match_id rename to idx_match_events_old_match_id;
alter index idx_match_events_created_at rename to idx_match_events_old_created_at;

create table match_events (
    id          bigint not null default nextval('match_events_id_seq'),
    match_id    bigint not null,
    player_id   bigint not null,
    event_type  text not null
                check (event_type in ('GOAL', 'ASSIST', 'YELLOW_CARD', 'RED_CARD', 'SUBSTITUTION')),
    minute      integer not null check (minute >= 0 and minute <= 130),
    created_at  timestamptz not null default now(),
    updated_at  timestamptz not null default now(),
    primary key (id, created_at)
) partition by range (created_at);

do $$
declare
    first_month date;
    last_month  date;
    m           date;
begin
    select least(
               date_trunc('month', min(created_at) at time zone 'utc'),
               date_trunc('month', now() at time zone 'utc') - interval '24 months'
           )::date
    into first_month
    from match_events_old;

    last_month := (date_trunc('month', now() at time zone 'utc') + interval '3 months')::date;
    m := first_month;

    while m <= last_month loop
        execute format(
            'create table %I partition of match_events for values from (%L) to (%L)',
            'match_events_' || to_char(m, 'YYYY_MM'),
            m || ' 00:00:00+00',
            (m + interval '1 month')::date || ' 00:00:00+00'
        );
        m := (m + interval '1 month')::date;
    end loop;
end $$;

insert into match_events (id, match_id, player_id, event_type, minute, created_at, updated_at)
select id, match_id, player_id, event_type, minute, created_at, updated_at
from match_events_old;

alter sequence match_events_id_seq owned by match_events.id;
drop table match_events_old;

alter table match_events
    add constraint match_events_match_id_fkey
    foreign key (match_id) references matches(id) on delete cascade;

alter table match_events
    add constraint match_events_player_id_fkey
    foreign key (player_id) references players(id) on delete cascade;

create index idx_match_events_player_created_at on match_events (player_id, created_at desc);
create index idx_match_events_match_id on match_events (match_id);
create index idx_match_events_created_at on match_events (created_at);

analyze match_events;
```

что тут важно:
* primary key теперь (id, created_at). постгрес не дает сделать уникальный ключ на партиционированной таблице если в него не входит ключ партиционирования
* партиции создаются циклом от месяца самого старого события (но не позже чем 24 месяца назад, чтобы на пустой базе генератор с `--days 730` не падал) до текущего месяца + 3 вперед
* данные переносятся insert select, а внешние ключи и индексы добавляются уже после переноса, так быстрее
* последовательность match_events_id_seq перевешивается на новую таблицу до drop старой, иначе drop удалил бы ее вместе со старой таблицей и новые события не смогли бы получить id
* индексы объявлены на родителе, постгрес сам создал их в каждой партиции
